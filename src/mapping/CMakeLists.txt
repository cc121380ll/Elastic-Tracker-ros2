cmake_minimum_required(VERSION 3.8)
project(mapping)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(quadrotor_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(OpenCV REQUIRED)
add_library(mapping src/mapping.cc)
target_include_directories(mapping PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS} ${pcl_conversions_INCLUDE_DIRS})
target_link_libraries(mapping PUBLIC Eigen3::Eigen rclcpp::rclcpp quadrotor_interfaces::quadrotor_interfaces__rosidl_typesupport_cpp sensor_msgs::sensor_msgs__rosidl_typesupport_cpp ${PCL_LIBRARIES})
add_library(mapping_component SHARED src/mapping_nodelet.cpp)
target_include_directories(mapping_component PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mapping_component PRIVATE mapping rclcpp_components::component nav_msgs::nav_msgs__rosidl_typesupport_cpp geometry_msgs::geometry_msgs__rosidl_typesupport_cpp cv_bridge::cv_bridge message_filters::message_filters ${OpenCV_LIBRARIES})
rclcpp_components_register_node(mapping_component PLUGIN "mapping::Nodelet" EXECUTABLE mapping_node)
add_executable(mapping_vis_node src/mapping_vis_node.cpp)
target_link_libraries(mapping_vis_node PRIVATE mapping)
set_target_properties(mapping PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS mapping EXPORT export_mapping ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS mapping_component LIBRARY DESTINATION lib)
install(TARGETS mapping_vis_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME} OPTIONAL)
ament_export_targets(export_mapping HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp quadrotor_interfaces sensor_msgs pcl_conversions Eigen3 PCL)
ament_package()
