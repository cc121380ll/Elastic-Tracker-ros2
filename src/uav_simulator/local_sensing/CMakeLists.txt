cmake_minimum_required(VERSION 3.18)
project(local_sensing_node LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 14)
# Preserve compute_86 / sm_86 for RTX 3060; CUDA is mandatory.
set(CMAKE_CUDA_ARCHITECTURES 86)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(OpenCV REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(Eigen3 REQUIRED)
find_package(CUDAToolkit 11.8 REQUIRED)
add_library(depth_render_cuda SHARED src/depth_render.cu)
target_include_directories(depth_render_cuda PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
target_compile_options(depth_render_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--use_fast_math>)
target_link_libraries(depth_render_cuda PUBLIC CUDA::cudart)
add_executable(pcl_render_node src/pcl_render_node.cpp)
target_include_directories(pcl_render_node SYSTEM PRIVATE ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${pcl_conversions_INCLUDE_DIRS})
target_link_libraries(pcl_render_node PRIVATE depth_render_cuda Eigen3::Eigen rclcpp::rclcpp
  sensor_msgs::sensor_msgs__rosidl_typesupport_cpp geometry_msgs::geometry_msgs__rosidl_typesupport_cpp
  nav_msgs::nav_msgs__rosidl_typesupport_cpp cv_bridge::cv_bridge ${pcl_conversions_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBS})
install(TARGETS pcl_render_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS depth_render_cuda LIBRARY DESTINATION lib)
install(DIRECTORY params/ DESTINATION share/${PROJECT_NAME}/params)
ament_package()
