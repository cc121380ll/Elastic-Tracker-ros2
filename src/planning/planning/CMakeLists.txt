cmake_minimum_required(VERSION 3.8)
project(planning)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(quadrotor_interfaces REQUIRED)
find_package(decomp_ros_utils REQUIRED)
find_package(traj_opt REQUIRED)
find_package(mapping REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
add_library(planning_component SHARED src/planning_nodelet.cpp)
target_include_directories(planning_component PRIVATE include ${PCL_INCLUDE_DIRS} ${pcl_conversions_INCLUDE_DIRS})
target_link_libraries(planning_component PRIVATE Eigen3::Eigen rclcpp::rclcpp rclcpp_components::component mapping::mapping traj_opt::traj_opt decomp_ros_utils::decomp_rviz_plugins ament_index_cpp::ament_index_cpp quadrotor_interfaces::quadrotor_interfaces__rosidl_typesupport_cpp std_msgs::std_msgs__rosidl_typesupport_cpp nav_msgs::nav_msgs__rosidl_typesupport_cpp geometry_msgs::geometry_msgs__rosidl_typesupport_cpp visualization_msgs::visualization_msgs__rosidl_typesupport_cpp ${PCL_LIBRARIES})
rclcpp_components_register_node(planning_component PLUGIN "planning::Nodelet" EXECUTABLE planning_node)
add_executable(traj_server src/traj_server.cpp)
target_include_directories(traj_server PRIVATE include)
target_link_libraries(traj_server PRIVATE traj_opt::traj_opt mapping::mapping quadrotor_interfaces::quadrotor_interfaces__rosidl_typesupport_cpp std_msgs::std_msgs__rosidl_typesupport_cpp nav_msgs::nav_msgs__rosidl_typesupport_cpp visualization_msgs::visualization_msgs__rosidl_typesupport_cpp)
install(TARGETS planning_component LIBRARY DESTINATION lib)
install(TARGETS traj_server DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME} OPTIONAL)
ament_package()
