cmake_minimum_required(VERSION 3.8)
project(decomp_ros_utils)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(decomp_ros_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Widgets)
set(CMAKE_AUTOMOC ON)
add_definitions(-DQT_NO_KEYWORDS)
add_library(decomp_rviz_plugins SHARED src/mesh_shape.cpp src/bound_visual.cpp src/mesh_visual.cpp src/vector_visual.cpp src/ellipsoid_array_visual.cpp src/ellipsoid_array_display.cpp src/polyhedron_array_display.cpp)
target_include_directories(decomp_rviz_plugins
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
         $<INSTALL_INTERFACE:include>
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/decomp_ros_utils)
target_link_libraries(decomp_rviz_plugins PUBLIC Eigen3::Eigen decomp_ros_interfaces::decomp_ros_interfaces__rosidl_typesupport_cpp sensor_msgs::sensor_msgs__rosidl_typesupport_cpp nav_msgs::nav_msgs__rosidl_typesupport_cpp rviz_common::rviz_common rviz_rendering::rviz_rendering pluginlib::pluginlib Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS decomp_rviz_plugins EXPORT export_decomp LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY icons DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE.upstream DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_decomp HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 decomp_ros_interfaces sensor_msgs nav_msgs rviz_common rviz_rendering pluginlib Qt5)
ament_package()
