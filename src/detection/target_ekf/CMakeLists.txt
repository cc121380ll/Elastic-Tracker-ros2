cmake_minimum_required(VERSION 3.8)
project(target_ekf)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(object_detection_interfaces REQUIRED)
find_package(message_filters REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(node target_ekf_node target_ekf_sim_node)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  target_link_libraries(${node} PRIVATE Eigen3::Eigen rclcpp::rclcpp message_filters::message_filters
    nav_msgs::nav_msgs__rosidl_typesupport_cpp
    object_detection_interfaces::object_detection_interfaces__rosidl_typesupport_cpp)
  install(TARGETS ${node} RUNTIME DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
ament_package()
